11/// Compress a [`JsString`].
12#[wasm_bindgen]
13pub fn compress(data: &JsValue) -> JsValue {
14    let data: &JsString = match data.dyn_ref::<JsString>() {
15        Some(data) => data,
16        None => {
17            return JsValue::NULL;
18        }
19    };
20    let data: Vec<u16> = data.iter().collect();
21    let compressed = lz_str::compress(data);
22    convert_u16_slice_to_string(&compressed).into()
23}
24
25/// Decompress a [`JsString`].
26#[wasm_bindgen]
27pub fn decompress(data: &JsValue) -> JsValue {
28    let data: &JsString = match data.dyn_ref::<JsString>() {
29        Some(data) => data,
30        None => {
31            return JsValue::NULL;
32        }
33    };
34    let data: Vec<u16> = data.iter().collect();
35    lz_str::decompress(data)
36        .map(|decompressed| convert_u16_slice_to_string(&decompressed).into())
37        .unwrap_or(JsValue::NULL)
38}
